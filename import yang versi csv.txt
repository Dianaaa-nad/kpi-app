def get_connection():
    return pymysql.connect(**DB_CONFIG, cursorclass=pymysql.cursors.DictCursor)

def load_lookup_data(conn):
    with conn.cursor() as cur:
        cur.execute("SELECT branch_id, branch_name FROM branches")
        branches = {r["branch_name"].strip().upper(): r["branch_id"] for r in cur.fetchall()}
        cur.execute("SELECT variable_id, var_code FROM kpi_variables")
        variables = {r["var_code"].strip().upper(): r["variable_id"] for r in cur.fetchall()}
    return branches, variables

def parse_periode(raw) -> date | None:
    if pd.isna(raw): return None
    s = str(raw).strip()
    for fmt in ("%Y-%m-%d", "%Y-%m", "%m/%Y", "%Y/%m", "%m-%Y"):
        try:
            return datetime.strptime(s[:len(fmt)+2], fmt).replace(day=1).date()
        except Exception:
            continue
    try:
        return pd.to_datetime(s).replace(day=1).date()
    except Exception:
        return None

def validate_rows(df: pd.DataFrame, branches: dict, variables: dict) -> pd.DataFrame:
    results = []
    for idx, row in df.iterrows():
        errors = []
        branch_name_raw = str(row.get("branch_name","")).strip()
        var_code_raw    = str(row.get("var_code","")).strip()
        branch_id   = branches.get(branch_name_raw.upper())
        variable_id = variables.get(var_code_raw.upper())

        if not branch_name_raw: errors.append("branch_name kosong")
        elif branch_id is None: errors.append(f"'{branch_name_raw}' tidak ditemukan")

        if not var_code_raw: errors.append("var_code kosong")
        elif variable_id is None: errors.append(f"var_code '{var_code_raw}' tidak ditemukan")

        parsed = parse_periode(row.get("periode"))
        if parsed is None: errors.append(f"periode '{row.get('periode')}' tidak valid")

        value = target = None
        try:
            v = row.get("value"); value = float(v) if pd.notna(v) and str(v).strip() != "" else None
        except: errors.append("value bukan angka")
        try:
            t = row.get("target"); target = float(t) if pd.notna(t) and str(t).strip() != "" else None
        except: errors.append("target bukan angka")

        results.append({
            "row_num": idx + 2,
            "branch_name": branch_name_raw, "var_code": var_code_raw,
            "periode_raw": str(row.get("periode","")), "value": value, "target": target,
            "branch_id": branch_id, "variable_id": variable_id, "parsed_periode": parsed,
            "status": "error" if errors else "ok",
            "error_message": "; ".join(errors) if errors else None,
        })
    return pd.DataFrame(results)

def upsert_realizations(conn, ok: pd.DataFrame) -> int:
    rows = ok[ok["value"].notna()]
    if rows.empty: return 0
    sql = """INSERT INTO kpi_realizations (branch_id,variable_id,periode,value)
             VALUES (%s,%s,%s,%s) ON DUPLICATE KEY UPDATE value=VALUES(value)"""
    data = [(r.branch_id, r.variable_id, r.parsed_periode, r.value) for r in rows.itertuples()]
    with conn.cursor() as cur: cur.executemany(sql, data)
    return len(data)

def upsert_targets(conn, ok: pd.DataFrame) -> int:
    rows = ok[ok["target"].notna()]
    if rows.empty: return 0
    sql = """INSERT INTO kpi_targets (branch_id,variable_id,periode,target_value)
             VALUES (%s,%s,%s,%s) ON DUPLICATE KEY UPDATE target_value=VALUES(target_value)"""
    data = [(r.branch_id, r.variable_id, r.parsed_periode, r.target) for r in rows.itertuples()]
    with conn.cursor() as cur: cur.executemany(sql, data)
    return len(data)

def save_log(conn, user_id, filename, status, notes, validated) -> int:
    with conn.cursor() as cur:
        cur.execute("INSERT INTO import_logs (imported_by,filename,status,notes) VALUES (%s,%s,%s,%s)",
                    (user_id, filename, status, notes))
        iid = cur.lastrowid
        cur.executemany(
            "INSERT INTO import_log_details (import_id,branch_name,var_code,value,periode,row_status,error_message) VALUES (%s,%s,%s,%s,%s,%s,%s)",
            [(iid, r.branch_name, r.var_code, r.value, r.parsed_periode, r.status, r.error_message)
             for r in validated.itertuples()])
    return iid

            <div class="metric-box metric-total"><div class="metric-num">{n_total}</div><div class="metric-label">Total Baris</div></div>
            <div class="metric-box metric-ok"><div class="metric-num">{n_ok}</div><div class="metric-label">✅ Valid</div></div>
            <div class="metric-box metric-err"><div class="metric-num">{n_err}</div><div class="metric-label">❌ Error</div></div>
        </div>
    """, unsafe_allow_html=True)

    if n_err > 0:
        st.markdown(f"""
        <div style="margin-bottom:16px;padding:14px 18px;background:var(--error-bg);border:1px solid var(--error-border);border-radius:10px;">
            <div style="font-size:13px;font-weight:700;color:var(--error);margin-bottom:10px;">❌ {n_err} baris bermasalah</div>
            <div class="preview-wrap">
            <table class="error-table">
                <thead><tr>
                    <th>Baris</th><th>Cabang</th><th>Var Code</th><th>Periode</th><th>Keterangan Error</th>
                </tr></thead>
                <tbody>
        """ + "".join([
            f'<tr class="row-error"><td>{r.row_num}</td><td>{r.branch_name}</td>'
            f'<td>{r.var_code}</td><td>{r.periode_raw}</td>'
            f'<td class="error-msg">{r.error_message}</td></tr>'
            for r in error_rows.itertuples()
        ]) + """
                </tbody>
            </table>
            </div>
        </div>
        """, unsafe_allow_html=True)

        err_csv = error_rows[["row_num","branch_name","var_code","periode_raw","value","target","error_message"]]\
                    .to_csv(index=False).encode("utf-8")
        st.download_button("⬇️ Download Laporan Error", data=err_csv,
                           file_name=f"error_{filename}", mime="text/csv")

    if n_ok == 0:
        st.markdown('<p style="color:var(--text-muted);font-size:13px">Tidak ada data valid untuk dilanjutkan.</p>', unsafe_allow_html=True)
        st.markdown("</div>", unsafe_allow_html=True)
        if st.button("← Ulangi Upload"): st.session_state.import_step = 1; st.rerun()
        st.stop()

    badge_cls  = "badge-success" if n_err == 0 else "badge-warning"
    badge_text = f"✓ {n_ok} baris siap diimport" if n_err == 0 else f"⚠️ {n_ok} baris valid, {n_err} dilewati"
    st.markdown(f'<span class="badge {badge_cls}">{badge_text}</span>', unsafe_allow_html=True)
    st.markdown("</div>", unsafe_allow_html=True)

    col_back, col_next = st.columns([1, 1])
    with col_back:
        if st.button("← Ulangi Upload", use_container_width=True):
            st.session_state.import_step = 1; st.rerun()
    with col_next:
        if st.button("➡️ Lihat Preview", type="primary", use_container_width=True):
            st.session_state.import_step = 3; st.rerun()


# ==============================================================================
# STEP 3 — PREVIEW & SUBMIT
# ==============================================================================
elif step == 3:
    validated  = st.session_state.validated_df
    filename   = st.session_state.get("_uploaded_file_name", "file.csv")

    if validated is None:
        st.session_state.import_step = 1; st.rerun()

    ok_rows    = validated[validated["status"] == "ok"].copy()
    error_rows = validated[validated["status"] == "error"]

    # Pivot preview: baris = branch+periode, kolom = var_code
    periods    = sorted(ok_rows["parsed_periode"].dropna().unique())
    var_codes  = sorted(ok_rows["var_code"].str.upper().unique())
    branches_u = sorted(ok_rows["branch_name"].unique())
    n_cells    = ok_rows["value"].notna().sum() + ok_rows["target"].notna().sum()

    periods_str = ", ".join([p.strftime("%b %Y") for p in periods])

    # Build pivot for value
    pivot = ok_rows.pivot_table(
        index=["branch_name","parsed_periode"],
        columns="var_code",
        values="value",
        aggfunc="first"
    ).reset_index()
    pivot.columns.name = None

    st.markdown(f"""
    <div class="import-card">
        <div style="display:flex;align-items:center;justify-content:space-between;margin-bottom:4px;">
            <div class="import-card-title">📊 Preview Data — {filename}</div>
            <span class="badge badge-success">✓ {len(ok_rows)} baris siap diimport</span>
        </div>
        <div class="import-card-desc">Data realisasi yang akan disimpan ke database (preview value)</div>
        <div class="info-strip">
            <div class="info-strip-item">📅 Periode: <span class="info-strip-value">{periods_str}</span></div>
            <div class="info-strip-item">🏦 Cabang: <span class="info-strip-value">{len(branches_u)}</span></div>
            <div class="info-strip-item">📋 Variabel: <span class="info-strip-value">{len(var_codes)}</span></div>
            <div class="info-strip-item">📊 Data Points: <span class="info-strip-value">{n_cells:,}</span></div>
        </div>
    """, unsafe_allow_html=True)

    # Render preview table HTML
    vc_cols  = [c for c in pivot.columns if c not in ("branch_name","parsed_periode")]
    th_heads = "".join([f"<th>{c}</th>" for c in vc_cols])
    rows_html = ""
    for _, row in pivot.iterrows():
        cells = "".join([
            f"<td>{'—' if pd.isna(row[c]) else f'{row[c]:,.0f}' if row[c] >= 1000 else f'{row[c]:g}'}</td>"
            for c in vc_cols
        ])
        period_label = row["parsed_periode"].strftime("%b %Y") if pd.notna(row["parsed_periode"]) else ""
        rows_html += f"<tr><td>{row['branch_name']}<br><small style='color:var(--text-muted);font-weight:400'>{period_label}</small></td>{cells}</tr>"

    st.markdown(f"""
    <div class="preview-wrap">
    <table class="preview-table">
        <thead><tr><th>Cabang / Periode</th>{th_heads}</tr></thead>
        <tbody>{rows_html}</tbody>
    </table>
    </div>
    """, unsafe_allow_html=True)

    if error_rows.shape[0] > 0:
        st.markdown(f"""
        <div style="margin-top:14px;padding:10px 16px;background:var(--warning-bg);border:1px solid var(--warning-border);border-radius:10px;font-size:12.5px;color:var(--warning-text);">
            ⚠️ <b>{error_rows.shape[0]} baris dilewati</b> karena error — tidak termasuk dalam import
        </div>
        """, unsafe_allow_html=True)

    st.markdown("</div>", unsafe_allow_html=True)

    col_back, col_spacer, col_submit = st.columns([1, 2, 1])
    with col_back:
        if st.button("← Kembali", use_container_width=True):
            st.session_state.import_step = 2; st.rerun()
    with col_submit:
        if st.button("✅ Submit Import", type="primary", use_container_width=True):
            user_id = st.session_state.get("user_id", 1)
            try:
                with st.spinner("Menyimpan ke database..."):
                    conn = get_connection()
                    n_val  = upsert_realizations(conn, ok_rows)
                    n_tgt  = upsert_targets(conn, ok_rows)
                    status = "success" if error_rows.empty else "partial"
                    notes  = f"{len(error_rows)} baris dilewati." if not error_rows.empty else None
                    iid    = save_log(conn, user_id, filename, status, notes, validated)
                    conn.commit()
                    conn.close()
                st.session_state.import_result = {
                    "n_val": n_val, "n_tgt": n_tgt, "iid": iid,
                    "n_err": len(error_rows), "periods": periods_str,
                }
                st.session_state.import_done = True
                st.session_state.import_step = 4
                st.rerun()
            except Exception as e:
                try: conn.rollback(); conn.close()
                except: pass
                st.error(f"Import gagal, semua perubahan dibatalkan.\n\nDetail: {e}")


# ==============================================================================
# STEP 4 — DONE
# ==============================================================================
elif step == 4:
    res = st.session_state.import_result

        </div>

            ⚠️ <b>{res['n_err']} baris dilewati</b> karena gagal validasi — silakan perbaiki dan upload ulang
        </div>
        """, unsafe_allow_html=True)

        validated = st.session_state.validated_df
        if validated is not None:
            err_rows = validated[validated["status"] == "error"]
            err_csv  = err_rows[["row_num","branch_name","var_code","periode_raw","value","target","error_message"]]\
                                .to_csv(index=False).encode("utf-8")
            st.download_button("⬇️ Download Laporan Error", data=err_csv,
                               file_name="error_import.csv", mime="text/csv")

    col1, col2 = st.columns(2)
    with col1:
        if st.button("📥 Import Data Baru", use_container_width=True, type="primary"):
            for k in ["import_step","validated_df","import_done","import_result","last_filename",
                      "_uploaded_file_bytes","_uploaded_file_name"]:
                st.session_state.pop(k, None)
            st.rerun()
    with col2:
        st.link_button("🏠 Kembali ke Dashboard", "/", use_container_width=True)
